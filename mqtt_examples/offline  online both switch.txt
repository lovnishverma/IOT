#include <ESP8266WiFi.h>
#include <PubSubClient.h>

const char* ssid = "wifiname";
const char* password = "yourwifipass";
const char* mqttServer = "public.mqtthq.com";
const int mqttPort = 1883;
const char* mqttTopic = "mqttHQ-client-topicname";
const int qos = 0;

WiFiServer server(80);
WiFiClient espClient;
PubSubClient client(espClient);

char lastMessage[2] = ""; // Variable to store the last received message
bool ledState = true; // Initial state of the LED is turned on

void setup() {
  pinMode(LED_BUILTIN, OUTPUT);
  digitalWrite(LED_BUILTIN, ledState ? LOW : HIGH); // Set LED state

  Serial.begin(115200);
  delay(10);

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.println("Connecting to WiFi...");
  }

  Serial.println("Connected to WiFi");
  Serial.print("IP address: ");
  Serial.println(WiFi.localIP());

  server.begin();

  client.setServer(mqttServer, mqttPort);
  client.setCallback(callback);

  while (!client.connected()) {
    Serial.println("Connecting to MQTT...");
    if (client.connect("ESP8266Client")) {
      Serial.println("Connected to MQTT");
      client.subscribe(mqttTopic, qos);

      // Clear retained message on MQTT broker
      client.publish(mqttTopic, "", true);
    } else {
      Serial.print("Failed, rc=");
      Serial.print(client.state());
      Serial.println(" Retrying in 5 seconds...");
      delay(5000);
    }
  }
}

void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();

  // Check for incoming client connections
  WiFiClient clientConnection = server.available();
  if (clientConnection) {
    processClient(clientConnection);
  }
}

void callback(char* topic, byte* payload, unsigned int length) {
  Serial.println("Message received: ");
  Serial.print((char*)payload);

  if (length == 1 && payload[0] != lastMessage[0]) {
    if (payload[0] == '0') {
      ledState = true; // Set LED state
    } else if (payload[0] == '1') {
      ledState = false; // Set LED state
    }

    digitalWrite(LED_BUILTIN, ledState ? LOW : HIGH); // Update LED state
    strncpy(lastMessage, (char*)payload, sizeof(lastMessage));
  }
}

void reconnect() {
  while (!client.connected()) {
    Serial.println("Connecting to MQTT...");
    if (client.connect("ESP8266Client")) {
      Serial.println("Connected to MQTT");
      client.subscribe(mqttTopic, qos);

      // Clear retained message on MQTT broker
      client.publish(mqttTopic, "", true);
    } else {
      Serial.print("Failed, rc=");
      Serial.print(client.state());
      Serial.println(" Retrying in 5 seconds...");
      delay(5000);
    }
  }
}

void processClient(WiFiClient client) {
  String request = client.readStringUntil('\r');
  client.flush();

  if (request.indexOf("/toggle") != -1) {
    ledState = !ledState; // Toggle LED state
    digitalWrite(LED_BUILTIN, ledState ? LOW : HIGH); // Update LED state
  }

  sendHTTPResponse(client);
  client.stop();
}

void sendHTTPResponse(WiFiClient client) {
  client.println("HTTP/1.1 200 OK");
  client.println("Content-Type: text/html");
  client.println("");
  client.println("<!DOCTYPE HTML>");
  client.println("<html>");
  client.println("<head><title>ESP8266 LED Control</title>");
  client.println("<style>");
  client.println("body { font-family: Arial, Helvetica, sans-serif;}");
  client.println("h1 { color: #333; }");
  client.println("p { margin-bottom: 20px; }");
  client.println(".button {");
  client.println("  display: inline-block;");
  client.println("  border-radius: 4px;");
  client.println("  background-color: #4CAF50;");
  client.println("  border: none;");
  client.println("  color: white;");
  client.println("  text-align: center;");
  client.println("  font-size: 16px;");
  client.println("  padding: 10px 20px;");
  client.println("  cursor: pointer;");
  client.println("  text-decoration: none;");
  client.println("}");
  client.println(".button:hover { background-color: #45a049; }");
  client.println("</style>");
  client.println("</head>");
  client.println("<body>");
  client.println("<h1>ESP8266 LED Control</h1>");
  client.println("<p>LED Status: " + String(ledState ? "ON" : "OFF") + "</p>");
  client.println("<a class=\"button\" href=\"/toggle\">Toggle LED</a>");
  client.println("</body></html>");
}
